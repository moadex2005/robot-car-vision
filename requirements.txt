# Install PyTorch first, from its own index, so it comes with graphics-card support:
#
#   .venv\Scripts\python.exe -m pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cu130
#
# Then install everything below:
#
#   .venv\Scripts\python.exe -m pip install -r requirements.txt
#
# Last, repair ONNX Runtime. InsightFace pulls in the processor-only build, which
# overwrites the graphics-card build, so remove both and put the right one back:
#
#   .venv\Scripts\python.exe -m pip uninstall -y onnxruntime onnxruntime-gpu
#   .venv\Scripts\python.exe -m pip install --no-deps onnxruntime-gpu==1.32.0
#
# check_env.py fails on "ONNX Runtime fell back to CPUExecutionProvider" if this step was missed.
